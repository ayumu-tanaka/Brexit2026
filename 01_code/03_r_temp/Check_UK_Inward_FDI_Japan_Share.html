Purpose

This temporary note calculates Japan’s share of inward FDI stocks in the United Kingdom using the UIFS4 bilateral inward FDI stock data.

The FDI stock values are interpreted as million USD, following Kox (2025). Missing bilateral cells are not interpreted as zeros; the denominator below is the sum of numerical UK inward FDI stock cells reported in the UIFS4 file for a given year.

The calculation is:

\[ \text{Japan share}_{t} = \frac{\text{inward FDI stock in GBR from JPN}_{t}} {\sum_p \text{inward FDI stock in GBR from partner } p_{t}} \times 100. \]

Calculation

uk_inward <- fdi %>%
  filter(iso3_r == "GBR")

uk_inward_total <- uk_inward %>%
  group_by(year) %>%
  summarise(
    uk_inward_fdi_total = sum(inw_fdi_stock, na.rm = TRUE),
    n_source_countries = n_distinct(iso3_p),
    n_nonmissing_sources = sum(!is.na(inw_fdi_stock)),
    .groups = "drop"
  )

uk_inward_from_japan <- uk_inward %>%
  filter(iso3_p == "JPN") %>%
  transmute(
    year,
    uk_inward_fdi_from_japan = inw_fdi_stock
  )

uk_japan_share <- uk_inward_total %>%
  left_join(uk_inward_from_japan, by = "year") %>%
  mutate(
    japan_share = uk_inward_fdi_from_japan / uk_inward_fdi_total,
    japan_share_percent = 100 * japan_share
  ) %>%
  arrange(year)

non_country_codes <- c("WLD", "WLG", "XLG")

uk_inward_ranking <- uk_inward %>%
  filter(
    !is.na(inw_fdi_stock),
    inw_fdi_stock > 0,
    !iso3_p %in% c("GBR", non_country_codes)
  ) %>%
  left_join(
    geo %>% select(iso3, source_country = ctry_name),

  group_by(year) %>%
  arrange(desc(inw_fdi_stock), .by_group = TRUE) %>%

  ungroup() %>%
  select(
    year,
    rank,
    iso3_p,
    source_country,
    uk_inward_fdi_stock = inw_fdi_stock
  )

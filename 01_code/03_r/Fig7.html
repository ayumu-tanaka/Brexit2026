2 バランスト・パネルの検証と作成

SDiD は \(N\) ユニット × \(T\) 期間のバランスト・パネルを前提とする（Arkhangelsky et al. 2021, §設定）。 退出した系列会社はその後の年に観測されないため、パネルが非バランストになる可能性がある。 全期間に観測されないユニットを除外してバランスト化する。

n_units <- length(unique(aff_highEUb$AffiliateCode))
n_years <- length(unique(aff_highEUb$year))
n_obs   <- nrow(aff_highEUb)
is_balanced <- (n_obs == n_units * n_years)

cat(sprintf("ユニット数（系列会社）: %d\n", n_units))
## ユニット数（系列会社）: 1694
cat(sprintf("期間数: %d\n", n_years))
## 期間数: 8
cat(sprintf("観測数: %d\n", n_obs))
## 観測数: 13552
cat(sprintf("バランスト・パネル: %s\n",
            if (is_balanced) "はい" else "いいえ（欠損あり）"))
## バランスト・パネル: はい
if (!is_balanced) {
  # 全期間（n_years 期間）に観測されるユニットのみ残す
  obs_count     <- table(aff_highEUb$AffiliateCode)
  complete_units <- names(obs_count[obs_count == n_years])
  n_dropped     <- n_units - length(complete_units)
  aff_highEUb   <- aff_highEUb[aff_highEUb$AffiliateCode %in% complete_units, ]
  cat(sprintf("全期間（%d 期間）に観測されないユニットを除外: %d ユニット削除\n",
              n_years, n_dropped))
  cat(sprintf("バランスト後: ユニット数 %d, 観測数 %d\n",
              length(complete_units), nrow(aff_highEUb)))
  # 再確認
  stopifnot(nrow(aff_highEUb) == length(complete_units) * n_years)
  cat("バランスト・パネル作成完了\n")
}

5 Bootstrap SE による処置効果の図示

Placebo 法は均一分散（homoskedasticity across units）を仮定するが、 バイナリ退出変数ではこの仮定が成り立ちにくい。 Bootstrap 法はユニット単位のクラスター・リサンプリングを行うため、 不均一分散・ユニット内系列相関に対してロバストである（Arkhangelsky et al. 2021, §I.C）。

# Bootstrap 法で標準誤差を計算（set.seed は直前に置く）
set.seed(12345)
se_boot <- sqrt(vcov(aff_highEUb.tau.hat, method = 'bootstrap'))
se_boot
##            [,1]
## [1,] 0.01549152
label1b <- sprintf('Point estimate = %1.3f', aff_highEUb.tau.hat)
label2b <- sprintf('S.E. (bootstrap) = %1.3f', se_boot)
label1b
## [1] "Point estimate = 0.016"
label2b
## [1] "S.E. (bootstrap) = 0.015"
sprintf('95%% CI (%1.2f, %1.2f)',
        aff_highEUb.tau.hat - 1.96 * se_boot,
        aff_highEUb.tau.hat + 1.96 * se_boot)
## [1] "95% CI (-0.01, 0.05)"
g1b <- plot(aff_highEUb.tau.hat, se.method = 'bootstrap')
g2b <- g1b +
  scale_x_continuous(breaks = ~ axisTicks(., log = FALSE)) +
  annotate("text", x = 2016, y = 0.1,  label = label1b, size = 3) +
  annotate("text", x = 2016, y = 0.05, label = label2b, size = 3)
